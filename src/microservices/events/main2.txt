package main

import (
    "encoding/json"
    "fmt"
    "log"
    "net/http"
    "context"

    "github.com/gorilla/mux"
    "github.com/segmentio/kafka-go"
)

type Event struct {
    Type    string `json:"type"`
    Payload string `json:"payload"`
}

type UserEvent struct {
    UserName string `json:"username"`
}

type PaymentEvent struct {
    Amount float64 `json:"amount"`
}

type MovieEvent struct {
    Title string `json:"title"`
}

func createProducer(brokers []string, topic string) (*kafka.Writer, error) {
    producer := &kafka.Writer{
        Addr:         kafka.TCP(brokers...),
        Topic:        topic,
        Balancer:     &kafka.LeastBytes{},
        RequiredAcks: kafka.RequireOne,
    }

    err := producer.Connect(context.Background())
    if err != nil {
        return nil, fmt.Errorf("failed to connect to Kafka: %w", err)
    }

    return producer, nil
}

func createConsumer(brokers []string, topic string, groupID string) (*kafka.Reader, error) {
    consumer := kafka.NewReader(kafka.ReaderConfig{
        Brokers:  brokers,
        Topic:    topic,
        GroupID:  groupID,
        MinBytes: 10e3,
        MaxBytes: 10e6,
    })

    err := consumer.Connect(context.Background())
    if err != nil {
        return nil, fmt.Errorf("failed to connect to Kafka: %w", err)
    }

    return consumer, nil
}

func consume(consumer *kafka.Reader) {
    for {
        msg, err := consumer.ReadMessage(context.Background())
        if err != nil {
            log.Printf("failed to read message: %v", err)
            continue
        }
        var event Event
        err = json.Unmarshal(msg.Value, &event)
        if err != nil {
            log.Printf("Failed to unmarshal event: %v", err)
            continue
        }
        log.Printf("Received event: %s with payload: %s", event.Type, event.Payload)
    }
}

func main() {
    router := mux.NewRouter()

    brokers := []string{"localhost:9092"}
    topic := "events"
    groupID := "event-group"

    producer, err := createProducer(brokers, topic)
    if err != nil {
        log.Fatalf("Failed to create producer: %v", err)
    }
    defer producer.Close()

    consumer, err := createConsumer(brokers, topic, groupID)
    if err != nil {
        log.Fatalf("Failed to create consumer: %v", err)
    }
    defer consumer.Close()

    go consume(consumer)

    router.HandleFunc("/event", func(w http.ResponseWriter, r *http.Request) {
        var event Event
        err := json.NewDecoder(r.Body).Decode(&event)
        if err != nil {
            http.Error(w, err.Error(), http.StatusBadRequest)
            return
        }

        body, err := json.Marshal(event)
        if err != nil {
            http.Error(w, err.Error(), http.StatusInternalServerError)
            return
        }

        err = producer.WriteMessages(context.Background(), kafka.Message{
            Key:   []byte(event.Type),
            Value: body,
        })
        if err != nil {
            http.Error(w, err.Error(), http.StatusInternalServerError)
            return
        }

        fmt.Fprintf(w, "Event created: %s", event.Type)
    }).Methods("POST")

    log.Fatal(http.ListenAndServe(":8082", router))
}
